import pandas as pd
import plotly.express as px

df = pd.read_sql("SELECT * FROM world_indicators", con)

# Scatter: GDP vs AI Adoption, animated by year
fig = px.scatter(df,
    x="GDP per capita",
    y="Estimated share ... generative AI",
    color="World region according to OWID",
    size="GDP per capita", size_max=30,
    animation_frame="Year",
    hover_name="Entity",
    template="simple_white"
)

# Box: Distribution by region
fig2 = px.box(df,
    x="World region according to OWID",
    y="Estimated share ... generative AI",
    color="World region according to OWID"
)
# Asia outliers: Singapore 60.9%, UAE 64%
fig.show()
Key insight: Higher GDP → higher AI adoption, but outliers like UAE (64%) and Singapore (60.9%) lead globally. Guyana vs Surinam shows income alone doesn't predict adoption.
